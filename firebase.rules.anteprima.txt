// REGOLE DI TRANSIZIONE — da usare mentre provi l'anteprima.
// Funzionano sia con la versione attuale in produzione sia con la nuova.
// Quando pubblichi la nuova versione, sostituiscile con firebase.rules.txt (più sicure).

// ---------- FIRESTORE ----------
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /vocali/{docId} {
      allow create: if request.resource.data.size is int && request.resource.data.size < 10 * 1024 * 1024;
      allow read, delete: if true;   // come oggi (la versione attuale non ha login)
      allow update: if request.resource.data.diff(resource.data).affectedKeys().hasOnly(['transcription']);
    }
    match /impostazioni/{docId} { allow read: if true; allow write: if request.auth != null; }
    match /opere/{docId}        { allow read: if true; allow write: if request.auth != null; }
  }
}

// ---------- STORAGE ----------
rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    match /vocali/{fileName} {
      allow create: if request.resource.size < 10 * 1024 * 1024 && request.resource.contentType.matches('audio/.*');
      allow read: if true;
      allow delete: if true;
    }
    match /impostazioni/{fileName} {
      allow read: if true;
      allow write: if request.auth != null && request.resource.size < 15 * 1024 * 1024 && request.resource.contentType.matches('image/.*');
    }
  }
}
